def tree(commit):
    rows = {}
    for entry in git('ls-tree', '-rz', commit).split(b'\0'):
        if not entry:
            continue
        meta, path = entry.split(b'\t', 1)
        mode, kind, oid = meta.decode().split()
        rows[path.decode()] = {'mode': mode, 'kind': kind, 'blob': oid}
    return rows

def blob_oid(data):
    return hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()

def root_tree_oid(entries):
    root = {}
    for path, entry in entries.items():
        d = root
        parts = path.split('/')
        for part in parts[:-1]:
            d = d.setdefault(part, {})
        d[parts[-1]] = (entry['mode'], entry['blob'])
    def hash_tree(d):
        rows = []
        for name, value in d.items():
            if isinstance(value, dict):
                rows.append((name + '/', '40000', name, hash_tree(value)))
            else:
                rows.append((name, value[0], name, value[1]))
        body = b''.join(mode.encode() + b' ' + name.encode() + b'\0' + bytes.fromhex(oid)
            for _, mode, name, oid in sorted(rows))
        return hashlib.sha1(b'tree ' + str(len(body)).encode() + b'\0' + body).hexdigest()
    return hash_tree(root)

    pos = 0
    index = 0
    while index < len(lines):
        match = re.match(rb'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@', lines[index])
        if not match:
            index += 1
            continue
        old_start = int(match[1])
        old_count = int(match[2]) if match[2] is not None else 1
        new_count = int(match[4]) if match[4] is not None else 1
        target = old_start - 1 if old_count else old_start
        assert target >= pos
        result.extend(original_lines[pos:target])
        pos = target
        used_old = used_new = 0
        index += 1
        while index < len(lines) and not lines[index].startswith(b'@@ '):
            line = lines[index]
            assert line[:1] in (b' ', b'+', b'-'), line

                used_old += 1
