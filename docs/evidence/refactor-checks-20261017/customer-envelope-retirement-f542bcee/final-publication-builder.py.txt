def put(name, data):
    assert pathlib.PurePosixPath(name).name == name and not name.endswith('.go')
    assert not data.startswith(b'\x7fELF')
    data.decode('utf-8')
    with (OUT / name).open('xb') as f:
        f.write(data)

def copy(path, name):
    path = pathlib.Path(path)
    assert path.is_file() and not path.is_symlink()
    data = path.read_bytes()
    put(name, data)
    copies.append({'origin': str(path), 'copy': name, 'sha256': sha(data), 'bytes': len(data), 'byte_exact': True})
