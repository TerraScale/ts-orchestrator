package wazevoapi

import (
	"fmt"
	"os"
	"strconv"
	"sync"
)

var PerfMap *Perfmap

func init() {
	if PerfMapEnabled {
		pid := os.Getpid()
		filename := "/tmp/perf-" + strconv.Itoa(pid) + ".map"

		fh, err := os.OpenFile(filename, os.O_APPEND|os.O_RDWR|os.O_CREATE, 0o644)
		if err != nil {
			panic(err)
		}

		PerfMap = &Perfmap{fh: fh}
	}
}

// Perfmap holds perfmap entries to be flushed into a perfmap file.
type Perfmap struct {
	entries []entry
	mux     sync.Mutex
	fh      *os.File
}

type entry struct {
	index  int
	offset int64
	size   uint64
	name   string
}
