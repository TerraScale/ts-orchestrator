def put(path, data):

def js(path, value):
    put(path, (json.dumps(value, indent=2, sort_keys=True) + '\n').encode())

def copy(source, dest):
    data = pathlib.Path(source).read_bytes()
    assert not data.startswith(b'\x7fELF')
    put(dest, data)
    bindings[dest] = {'origin': str(source), 'sha256': sha(data), 'byte_exact_copy': True}

def tree(c):
    result = {}
    for row in git('ls-tree', '-r', '-z', c).split(b'\0'):
        if not row:
            continue
        meta, path = row.split(b'\t', 1)
        mode, kind, oid = meta.decode().split()
        result[path.decode()] = {'mode': mode, 'kind': kind, 'blob': oid}
    return result
