import fcntl, hashlib, json, pathlib, resource, subprocess, sys, time
OUT = pathlib.Path(__file__).parent
REQUEST = pathlib.Path('/tmp/ts-attempt-cc37-shortb-source-nar2velj/request.json')
EXPECTED_REQUEST_SHA = 'f157b41fa9029e661e2bb33b4a9bc508c81cfebe992cb86c8c914665ec5815de'
def sha(data): return hashlib.sha256(data).hexdigest()
def write(name, value):
    with (OUT / name).open('x') as f:
        json.dump(value, f, indent=2)
        f.write('\n')
def git(w, *args):
    return subprocess.check_output(['/run/current-system/sw/bin/git', '-C', str(w), *args])
raw = REQUEST.read_bytes()
assert sha(raw) == EXPECTED_REQUEST_SHA
request = json.loads(raw)
w = pathlib.Path(request['worktree'])
assert len(sys.argv) == 1 and sys.flags.isolated and sys.flags.dont_write_bytecode
assert str(pathlib.Path(sys.executable).resolve()) == str(pathlib.Path(request['executable_paths']['python']).resolve())
def snapshot():
    status = git(w, 'status', '--porcelain=v1', '--untracked-files=all').decode()
    return {'head': git(w, 'rev-parse', 'HEAD').decode().strip(),
            'tree': git(w, 'rev-parse', 'HEAD^{tree}').decode().strip(),
            'status': status, 'status_sha256': sha(status.encode()),
            'source_sha256': sha((w / request['owned_changed_paths'][0]).read_bytes()),
            'original_sha256': sha((w / request['retained_original_path']).read_bytes()),
            'module_sha256': {p: sha((w / p).read_bytes()) for p in request['module_sha256']},
            'executable_sha256': {k: sha(pathlib.Path(p).read_bytes()) for k, p in request['executable_paths'].items()},
            'request_sha256': sha(REQUEST.read_bytes()),
            'runner_sha256': sha(pathlib.Path(__file__).read_bytes())}
pre = snapshot()
assert pre['head'] == request['commit'] and pre['tree'] == request['tree'] and not pre['status']
assert pre['source_sha256'] == request['source_file_sha256']
assert pre['original_sha256'] == request['retained_original_file_sha256']
assert pre['module_sha256'] == request['module_sha256']
assert pre['executable_sha256'] == request['executable_sha256']
assert pre['request_sha256'] == EXPECTED_REQUEST_SHA
write('attempt.json', {'state': 'ONE_SHOT_WRAPPER_ENTERED', 'source': pre})
with open(request['request']['shared_lock'], 'a') as gate:
    try:
        fcntl.flock(gate, fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError:
        write('receipt.json', {'state': 'LOCK_BUSY_NOT_LAUNCHED', 'source': pre, 'retry_authorized': False})
        raise SystemExit(75)
    assert snapshot() == pre
    write('preflight.json', {'source': pre, 'argv': request['request']['argv'], 'cwd': str(w),
                            'effective_child_environment': request['effective_child_environment'], 'GOROOT_present': False,
                            'python_flags': {'isolated': sys.flags.isolated, 'dont_write_bytecode': sys.flags.dont_write_bytecode}})
    started = time.time()
    clock = time.monotonic()
    with (OUT / 'stdout.jsonl').open('xb') as stdout, (OUT / 'stderr.log').open('xb') as stderr:
        proc = subprocess.Popen(request['request']['argv'], cwd=w, stdout=stdout, stderr=stderr, start_new_session=True)
        write('started.json', {'pid': proc.pid, 'epoch': started, 'argv': request['request']['argv']})
        print(json.dumps({'state': 'STARTED', 'pid': proc.pid, 'output': str(OUT)}), flush=True)
        returncode = proc.wait()
    elapsed = time.monotonic() - clock
    post = snapshot()
    actions, packages, run_tests, all_test_names, malformed = [], [], [], set(), []
    for number, line in enumerate((OUT / 'stdout.jsonl').read_text().splitlines(), 1):
        try:
            event = json.loads(line)
        except json.JSONDecodeError:
            malformed.append(number)
            continue
        if event.get('Test'):
            all_test_names.add(event['Test'])
            if event.get('Action') == 'run': run_tests.append(event['Test'])
            if event.get('Action') in ['pass', 'fail', 'skip']:
                actions.append({key: event[key] for key in ['Package', 'Test', 'Action', 'Elapsed'] if key in event})
        elif event.get('Action') in ['pass', 'fail', 'skip']:
            packages.append(event)
    expected = request['expected_pass_tests']
    success = (returncode == 0 and not malformed and run_tests == expected and all_test_names == set(expected)
               and len(actions) == 1 and actions[0]['Test'] == expected[0] and actions[0]['Action'] == 'pass'
               and actions[0].get('Package') == request['expected_package'] and len(packages) == 1
               and packages[0].get('Package') == request['expected_package'] and packages[0]['Action'] == 'pass'
               and pre == post)
    receipt = {'state': 'PASS' if success else 'NOT_QUALIFIED', 'exit_code': returncode, 'wall_seconds': elapsed,
               'max_child_rss_kib': resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss,
               'source_pre': pre, 'source_post': post, 'expected_tests': expected, 'actual_run_tests': run_tests,
               'actual_test_names': sorted(all_test_names), 'test_outcomes': actions, 'package_outcomes': packages,
               'malformed_stdout_lines': malformed, 'stdout_sha256': sha((OUT / 'stdout.jsonl').read_bytes()),
               'stderr_sha256': sha((OUT / 'stderr.log').read_bytes()), 'attempts': 1, 'retry_authorized': False,
               'scope': 'one nonrace actual new-root attempt; no child leaves, original retirement or global closure'}
    write('receipt.json', receipt)
    with (OUT / 'SHA256SUMS').open('x') as f:
        for path in sorted(OUT.iterdir()):
            if path.is_file() and path.name != 'SHA256SUMS':
                f.write(sha(path.read_bytes()) + '  ' + path.name + '\n')
    print(json.dumps(receipt), flush=True)
    fcntl.flock(gate, fcntl.LOCK_UN)
raise SystemExit(0 if success else 1)
