def put(name, data, origin, role, qualification, copied=False):
    assert '/' not in name and '\\' not in name and not name.endswith('.go')
    assert not data.startswith(b'\x7fELF') and b'\0' not in data
    data.decode('utf-8')
    with (OUT / name).open('xb') as f:
        f.write(data)
    entries[name] = {'origin': str(origin), 'sha256': sha(data), 'bytes': len(data), 'role': role, 'qualification': qualification, 'byte_exact_copy': copied}

def js(name, obj, role='publication source proof'):
    put(name, encoded(obj), 'derived read-only in this packet', role, 'PREPARED_ONLY_NO_ADOPTION_OR_EXECUTION')

def safe_name(path):
    result = path.as_posix().replace('/', '--')
    return result + '.txt' if result.endswith(('.go', '.py', '.sh')) else result

def copy_tree(directory, prefix, qualification, recursive=True, omit_dirs=()):
    paths = sorted(directory.rglob('*') if recursive else directory.iterdir())
    for p in paths:
        rel = p.relative_to(directory)
        if any(part in omit_dirs for part in rel.parts):
            if p.is_file():
                exclusions.append({'origin': str(p), 'reason': 'ELF output directory excluded; binary metadata retained in receipts and excluded-binaries.json'})
            continue
        assert not p.is_symlink(), p
        if not p.is_file():
            continue
        assert stat.S_ISREG(p.stat().st_mode)
        put(prefix + '--' + safe_name(rel), p.read_bytes(), p, prefix, qualification, True)

def tree(c):
    d = {}
    for row in git('ls-tree', '-r', '-z', c).split(b'\0'):
        if row:
            meta, name = row.split(b'\t', 1)
            mode, kind, oid = meta.decode().split()
            assert kind == 'blob', (c, name, kind)
            d[name.decode()] = {'mode': mode, 'blob': oid}
    return d

        elif line.startswith('-'):
            assert source[pos] == line[1:]; pos += 1
