import hashlib
import json
import pathlib
import subprocess
import tempfile

OLD = pathlib.Path('/tmp/ts-attempt-b3e1-truecompile-shortb-sealed-5di2rygp')
COMPAT = pathlib.Path('/tmp/ts-attempt-b8-narrow-compatibility-14x9e7qk')
PREP = pathlib.Path('/tmp/ts-attempt-retirement-b8-gsewmzm1')
W = PREP / 'source'
OUT = pathlib.Path(tempfile.mkdtemp(prefix='ts-attempt-458ff55-truecompile-shortb-sealed-'))
sha = lambda b: hashlib.sha256(b).hexdigest()
encoded = lambda v: (json.dumps(v, indent=2) + '\n').encode()
git = lambda *a: subprocess.check_output(['/run/current-system/sw/bin/git', '-C', str(W), *a])
bindings = {}

def put(n, b):
    with (OUT / n).open('xb') as f:
        f.write(b)

def copy(src, name):
    p = pathlib.Path(src)
    b = p.read_bytes()
    assert not b.startswith(b'\x7fELF')
    put(name, b)
    bindings[name] = {'origin': str(p), 'sha256': sha(b), 'copy_byte_exact': True}

proof = json.loads((PREP / 'source-preparation.json').read_text())
commit, tree, base = proof['commit'], proof['tree'], proof['base']

assert tree == 'e48c102cf49f21d338a09050d60854217cc70f2d'
assert git('rev-parse', 'HEAD').decode().strip() == commit
assert git('rev-parse', 'HEAD^{tree}').decode().strip() == tree
assert git('rev-parse', 'HEAD^').decode().strip() == base
assert not git('status', '--porcelain=v1', '--untracked-files=all')
paths = git('ls-files', '-z').decode().strip('\0').split('\0')
manifest = [{'path': p, 'sha256': sha((W / p).read_bytes())} for p in paths]
put('fullsource-manifest.json', encoded(manifest))

copy(COMPAT / 'seal.json', 'feature-b8-compatibility-seal.json')

copy(OLD / 'independent-source-review.json', 'historical-independent-source-review.json')
for name in ['accepted-runtime-review.json', 'accepted-runtime-receipt.json', 'accepted-runtime-stdout.jsonl', 'accepted-runtime-stderr.log', 'accepted-runtime-SHA256SUMS.txt', 'actual-prior-launch-tool-result.json', 'actual-prior-terminal-tool-result.json', 'actual-prior-native-SHA256SUMS.txt']:
    copy(OLD / name, name)
copy('/tmp/ts-qa-attempt-b3e1-truecompile-terminal-e9gh2dqb/review.json', 'historical-b3e1-compile-acceptance.json')
copy(OLD / 'qa-output/receipt.json', 'historical-b3e1-compile-receipt.json')
copy(__file__, 'packet-builder.py.txt')
actual_patch = git('diff', base, commit, '--', 'internal/attempt/cloud_test.go', 'internal/attempt/phase_dispatch_integration_test.go')
put('candidate.patch', actual_patch)
source_proof = {
    **proof,
    'tracked_files': len(manifest),
    'all_tracked_source_manifest_sha256': sha(encoded(manifest)),
    'all_other_paths_preserved': git('diff', '--name-status', base, commit).decode().splitlines() == proof['changed_paths'],
    'minimal_deletion_bytes': 253,
    'minimal_deletion_lines': 7,
    'removed_sha256': 'fbfd1c9f6466ea7c7cb2d93200e21c6c2e2159e4e5d10db02a4b1e01a392f6a1',
    'root_git_blob': git('rev-parse', commit + ':internal/attempt/phase_dispatch_integration_test.go').decode().strip(),
    'connected_sources': {p: sha((W / p).read_bytes()) for p in ['internal/attempt/attempt.go', 'internal/attempt/attempt_test.go', 'internal/entityid/id.go', 'internal/cli/cli.go', 'third_party/nats-server/server/accounts.go', 'third_party/nats-server/server/server.go']},
    'historical_runtime': {'commit': 'cc37fe13cace70584c45ee368d6edcd80ebc5f2d', 'root': 'TestRetiredContinuityPhaseThroughAttemptRun', 'acceptance_sha256': sha((OUT / 'accepted-runtime-review.json').read_bytes()), 'new_runtime_authorized': False},
