            category, reason = 'local-grpc-registration', 'Assigns function values or registers codecs/resolvers/balancers; parses an empty service config or constructs buffer pools. No Dial/Listen; callback bodies are deferred.'

            category, reason = 'unresolved-init', 'Not assigned a reviewed effect family; do not infer safe execution.'
    elif set(item['calls_including_literals_and_conversions']) <= benign:
        category, reason = 'local-construction-candidate', 'Lexical calls are known local allocation/error/regex/reflection/descriptor constructors or conversions. No whole-program effect proof claimed.'
    else:
        category, reason = 'unresolved-var', 'Call candidate requires further callee review; no safe-execution conclusion.'

        ownership = 'first-party-repository'
    elif item['module'] == 'terrascale.tech/sdk':
        ownership = 'first-party-sdk-cache'

        ownership = 'third-party-local-replacement'
    else:
        ownership = 'third-party-cache'
