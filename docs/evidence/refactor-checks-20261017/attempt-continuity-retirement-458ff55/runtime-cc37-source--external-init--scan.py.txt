def tag(name):

    if name.startswith('go1.'):

def combine(a, b, op):
    if op == '&&': return False if a is False or b is False else (True if a is True and b is True else None)
    return True if a is True or b is True else (False if a is False and b is False else None)
def evaluate(expr):
    terms = re.findall(r'&&|\|\||!|\(|\)|[A-Za-z0-9_.]+', expr); pos = 0
    def unary():
        nonlocal pos
        tok = terms[pos]; pos += 1
        if tok == '!':
            value = unary(); return None if value is None else not value
        if tok == '(':
            value = either(); assert terms[pos] == ')'; pos += 1; return value
        return tag(tok)
    def both():
        nonlocal pos
        value = unary()
        while pos < len(terms) and terms[pos] == '&&':
            pos += 1; value = combine(value, unary(), '&&')
        return value
    def either():
        nonlocal pos
        value = both()
        while pos < len(terms) and terms[pos] == '||':
            pos += 1; value = combine(value, both(), '||')
        return value
    value = either(); assert pos == len(terms); return value
def selection(path, src):
    stem = path.stem
    if stem.endswith('_test'): stem = stem[:-5]
    suffix = stem.split('_')
    if suffix[-1] in known_arch and suffix[-1] != 'amd64': return 'excluded-filename', ''
    if suffix[-1] in known_os and suffix[-1] != 'linux': return 'excluded-filename', ''
    if len(suffix) > 1 and suffix[-1] in known_arch and suffix[-2] in known_os and suffix[-2] != 'linux': return 'excluded-filename', ''
    constraint = re.search(r'(?m)^//go:build (.+)$', src)
    if constraint:
        try: value = evaluate(constraint[1])
        except Exception: return 'ambiguous-constraint', constraint[1]
        return {True: 'selected-static', False: 'excluded-constraint', None: 'ambiguous-cgo-or-experiment'}[value], constraint[1]
    if re.search(r'(?m)^//\s*\+build\s+', src): return 'ambiguous-legacy-constraint', ''
    return 'selected-static', ''
def imported(src):
    # Import declarations precede ordinary declarations; scanning their text avoids tokenizing generated OCI models.
    blocks = re.findall(r'(?m)^import\s*\((.*?)^\)', src, re.S) + re.findall(r'(?m)^import\s+((?:[.\w]+\s+)?"[^"\n]+")', src)
    return set(name for block in blocks for name in re.findall(r'"([^"\n]+)"', block))
def declarations(src):
    ts = tokens(src); result = []; depth = 0; i = 0
    while i < len(ts):
        tok = ts[i]
        if depth == 0 and tok[0] == 'func' and i + 2 < len(ts) and ts[i + 1][0] in ['init', 'TestMain']:
            j = paired(ts, i + 2, '(', ')') + 1
            while ts[j][0] != '{': j += 1
            end = paired(ts, j, '{', '}')
            result.append({'kind': ts[i + 1][0], 'line': tok[3], 'source': src[tok[1]:ts[end][2]]})
            i = end + 1; continue
        if depth == 0 and tok[0] == 'var':
            j = i + 1; nesting = []; end = j
            while j < len(ts):
                value = ts[j][0]
                if value in ['(', '[', '{']: nesting.append(value)
                elif value in [')', ']', '}']:
                    assert nesting and nesting.pop() == {')': '(', ']': '[', '}': '{'}[value]
                end = j
                if not nesting and (value == ';' or j + 1 == len(ts)): break
                if not nesting and ts[j + 1][3] > ts[j][3] and (ts[j][4] in ['identifier', 'literal'] or value in [')', ']', '}'] or value.isdigit()): break
                j += 1
            text = src[tok[1]:ts[end][2]]
            calls = set()
            for k in range(i + 1, end):
                if ts[k][4] != 'identifier' or ts[k - 1][0] == '.': continue
                name = ts[k][0]; next_token = k + 1
                while next_token + 1 <= end and ts[next_token][0] == '.' and ts[next_token + 1][4] == 'identifier':
                    name += '.' + ts[next_token + 1][0]; next_token += 2
                if next_token <= end and ts[next_token][0] == '(' and name not in ['var', 'func']: calls.add(name)
            calls = sorted(calls)
            if calls: result.append({'kind': 'package-var-candidate', 'line': tok[3], 'source': text, 'calls_including_literals_and_conversions': calls})
            i = end + 1; continue
        if tok[4] != 'literal':
            if tok[0] == '{': depth += 1
            elif tok[0] == '}': depth -= 1
        i += 1
    return result
