def parse_mod(path):
    text = path.read_text()
    name = re.search(r'^module\s+(\S+)', text, re.M).group(1)
    requires, replacements = {}, {}
    section = None
    for line in text.splitlines():
        line = line.split('//',1)[0].strip()
        if not line: continue
        if line in ('require (', 'replace ('):
            section = line.split()[0]; continue
        if line == ')': section = None; continue
        if line.startswith('require '): kind, value = 'require', line[8:]
        elif line.startswith('replace '): kind, value = 'replace', line[8:]
        else: kind, value = section, line
        if kind == 'require':
            parts = value.split()
            if len(parts) >= 2: requires[parts[0]] = parts[1]
        elif kind == 'replace' and '=>' in value:
            before, after = value.split('=>',1)
            replacements[before.split()[0]] = after.split()
    return name, requires, replacements

def build_possible(expr):
    ts = re.findall(r'\|\||&&|!|\(|\)|[A-Za-z0-9_.]+', expr); position = 0
    def atom():
        nonlocal position
        token = ts[position]; position += 1
        if token == '!': return {not x for x in atom()}
        if token == '(':
            result = disjunction(); assert ts[position] == ')'; position += 1; return result
        return tag_value(token)
    def conjunction():
        nonlocal position
        result = atom()
        while position < len(ts) and ts[position] == '&&':
            position += 1; rhs = atom(); result = {a and b for a in result for b in rhs}
        return result
    def disjunction():
        nonlocal position
        result = conjunction()
        while position < len(ts) and ts[position] == '||':
            position += 1; rhs = conjunction(); result = {a or b for a in result for b in rhs}
        return result
    result = disjunction(); assert position == len(ts)
    return result

def file_selection(path, source):
    stem = path.name[:-3]
    if stem.endswith('_test'): stem = stem[:-5]
    parts = stem.split('_')
    if path.name.startswith(('.', '_')): return False, 'ignored filename'
    for part in parts[-2:]:
        if part in OS and part != 'linux': return False, 'non-linux filename'
        if part in ARCH and part != 'amd64': return False, 'non-amd64 filename'
    match = re.search(r'^//go:build\s+(.+)$',source,re.M)
    if match:
        try:
            possibilities = build_possible(match.group(1))
            return True in possibilities, 'build constraint '+match.group(1)+(' [overapprox]' if len(possibilities)>1 else '')
        except (AssertionError,IndexError) as exc:
            errors.append({'file':str(path),'error':'build expression: '+str(exc)})
            return True, 'unresolved build constraint [overapprox]'
    legacy = re.findall(r'^// \+build (.+)$',source,re.M)
    return True, 'legacy-only build constraints [overapprox]' if legacy else 'linux-amd64 filename/default'

def imports_of(ts):
    found = []
    index = 0
    while index < len(ts):
        if ts[index][0] in ('func','var','type','const'): break
        if ts[index][0] == 'import':
            end = paired(ts,index+1,'(',')') if ts[index+1][0] == '(' else index+2
            for token in ts[index+1:end+1]:
                if token[4] == 'literal' and token[0].startswith(('"','`')):
                    value = token[0][1:-1]
                    if value not in found: found.append(value)
            index = end
        index += 1
    return found

def calls_of(ts):
    found = set()
    for index, token in enumerate(ts):
        if token[4] != 'identifier': continue
        if index and ts[index-1][0] == '.': continue
        end = index+1; parts = [token[0]]
        while end+1 < len(ts) and ts[end][0] == '.' and ts[end+1][4] == 'identifier':
            parts.append(ts[end+1][0]); end += 2
        if end < len(ts) and ts[end][0] == '(' and parts[0] not in ('if','for','switch','func','select'):
            found.add('.'.join(parts))
    return sorted(found)
