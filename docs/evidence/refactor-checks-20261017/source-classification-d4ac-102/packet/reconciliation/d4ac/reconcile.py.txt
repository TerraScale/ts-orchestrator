def js(name, value):
    put(name, (json.dumps(value, indent=2, sort_keys=True) + '\n').encode())

def rows(data):
    return list(csv.DictReader(io.StringIO(data.decode()), delimiter='\t'))

def tsv(name, values, columns=None):
    stream = io.StringIO()
    writer = csv.DictWriter(stream, fieldnames=columns or list(values[0]), delimiter='\t', lineterminator='\n')
    writer.writeheader()
    writer.writerows(values)
    put(name, stream.getvalue().encode())

def read_input(path):
    data = pathlib.Path(path).read_bytes()
    inputs[str(path)] = {'sha256': sha(data), 'bytes': len(data)}
    return data

def tree(commit):
    result = {}
    for entry in git('ls-tree', '-rz', commit).split(b'\0'):
        if entry:
            meta, path = entry.split(b'\t', 1)
            mode, kind, oid = meta.decode().split()
            result[path.decode()] = {'mode': mode, 'kind': kind, 'blob': oid}
    return result

def batch(oids):
    stream = io.BytesIO(git('cat-file', '--batch', data=''.join(oid + '\n' for oid in oids).encode()))
    for expected in oids:
        oid, kind, size = stream.readline().decode().split()
        assert oid == expected and kind == 'blob'
        data = stream.read(int(size))
        assert stream.read(1) == b'\n'
        yield data
    assert not stream.read()
