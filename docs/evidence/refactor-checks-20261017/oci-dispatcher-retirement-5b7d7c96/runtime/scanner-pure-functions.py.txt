def sha(data):
    return hashlib.sha256(data).hexdigest()

def letter(c):
    return c == '_' or unicodedata.category(c).startswith('L')

def tokens(src):
    # Offsets are Unicode character offsets, not byte offsets; hashes use re-encoded UTF-8.
    result = []
    i, line = 0, 1
    while i < len(src):
        c, start, ln = src[i], i, line
        if c.isspace():
            line += c == '\n'
            i += 1
            continue
        if src.startswith('//', i):
            j = src.find('\n', i + 2)
            i = len(src) if j < 0 else j
            continue
        if src.startswith('/*', i):
            j = src.find('*/', i + 2)
            if j < 0:
                raise ValueError('unterminated block comment')
            i = j + 2
            line += src[start:i].count('\n')
            continue
        if c in '\"\'`':
            quote = c
            i += 1
            while i < len(src):
                if src[i] == quote:
                    i += 1
                    break
                if quote != '`' and src[i] == '\\':
                    i += 2
                else:
                    i += 1
            else:
                raise ValueError('unterminated string/rune literal')
            line += src[start:i].count('\n')
            kind = 'literal'
        elif letter(c):
            i += 1
            while i < len(src) and (letter(src[i]) or unicodedata.category(src[i]) == 'Nd'):
                i += 1
            kind = 'identifier'
        else:
            i += 1
            kind = 'punctuation'
        result.append((src[start:i], start, i, ln, kind))
    return result

def paired(ts, start, left, right):
    depth = 0
    for j in range(start, len(ts)):
        if ts[j][4] == 'literal':
            continue
        if ts[j][0] == left:
            depth += 1
        elif ts[j][0] == right:
            depth -= 1
            if depth == 0:
                return j
    raise ValueError('unbalanced ' + left)

def imports(ts, path):
    aliases = []
    depth = 0
    for i, token in enumerate(ts):
        value = token[0]
        if value == '{': depth += 1
        if value == '}': depth -= 1
        if depth or value != 'import': continue
        j = i + 1
        end = paired(ts, j, '(', ')') if ts[j][0] == '(' else min(len(ts), j + 2)
        for k in range(j, end + 1):
            if k >= len(ts): break
            if ts[k][0] in ('"' + path + '"', '`' + path + '`'):
                previous = ts[k - 1]
                aliases.append(previous[0] if (previous[4] == 'identifier' and previous[0] != 'import') or previous[0] == '.' else path.rsplit('/', 1)[-1])
    return aliases

def entry_kind(name):
    if name == 'TestMain': return 'testmain'
    for prefix, kind in [('Test', 'test'), ('Benchmark', 'benchmark'), ('Fuzz', 'fuzz'), ('Example', 'example')]:
        if name.startswith(prefix) and (name == prefix or unicodedata.category(name[len(prefix)]) != 'Ll'):
            return kind
    return 'helper'

def extract(src):
    ts = tokens(src)
    testing_aliases = imports(ts, 'testing')
    tier_aliases = imports(ts, MODULE + '/internal/testutil/tier')
    result = []
    depth, i = 0, 0
    while i < len(ts):
        token = ts[i]
        if depth == 0 and token[0] == 'func' and i + 2 < len(ts) and ts[i + 1][4] == 'identifier':
            name = ts[i + 1][0]
            j = i + 2
            generic = ts[j][0] == '['
            if generic: j = paired(ts, j, '[', ']') + 1
            if ts[j][0] != '(':
                raise ValueError('function without parameter list at line ' + str(token[3]))
            params_end = paired(ts, j, '(', ')')
            body_start = params_end + 1
            while body_start < len(ts):
                value = ts[body_start][0]
                if value in ('(', '['):
                    body_start = paired(ts, body_start, value, ')' if value == '(' else ']') + 1
                    continue
                if value == '{' and ts[body_start - 1][0] in ('struct', 'interface'):
                    body_start = paired(ts, body_start, '{', '}') + 1
                    continue
                if value in ('{', ';', 'var', 'const', 'type'):
                    break
                if value == 'func' and ts[body_start + 1][0] != '(':
                    break
                body_start += 1
            has_body = body_start < len(ts) and ts[body_start][0] == '{'
            end = paired(ts, body_start, '{', '}') if has_body else params_end
            kind = entry_kind(name)
            param_tokens = [t[0] for t in ts[j + 1:params_end]]
            if param_tokens[-1:] == [',']: param_tokens.pop()
            expected_type = {'test': 'T', 'benchmark': 'B', 'fuzz': 'F', 'testmain': 'M'}.get(kind)
            type_options = [['*', alias, '.', expected_type] if alias != '.' else ['*', expected_type]
                            for alias in testing_aliases if alias != '_'] if expected_type else []
            valid_param = any(param_tokens == option or (len(param_tokens) == len(option) + 1 and letter(param_tokens[0][0]) and param_tokens[1:] == option) for option in type_options)
            if kind == 'example': valid_param = not param_tokens
            signature = 'not-an-entry-candidate' if kind == 'helper' else ('lexically-compatible-not-typechecked' if valid_param and not generic and body_start == params_end + 1 and has_body else 'unknown-or-incompatible-not-typechecked')
            gates = []
            body = ts[body_start + 1:end] if has_body else []
            for k, tok in enumerate(body):
                for alias in tier_aliases:
                    if alias == '.':
                        if tok[0] in ('Integration', 'Qualification', 'Unavailable') and k + 1 < len(body) and body[k + 1][0] == '(':
                            gates.append(tok[0] + '@' + str(tok[3]) + ':dot-import-ambiguous')
                    elif alias != '_' and tok[0] == alias and k + 3 < len(body) and body[k + 1][0] == '.' and body[k + 2][0] in ('Integration', 'Qualification', 'Unavailable') and body[k + 3][0] == '(':
                        gates.append(body[k + 2][0] + '@' + str(tok[3]))
            result.append(dict(name=name, kind=kind, line=token[3], end_line=ts[end][3],
                               declaration_sha256=sha(src[token[1]:ts[end][2]].encode()),
                               signature_status=signature, direct_tier_calls=';'.join(gates),
                               classification='unclassified', review_closure='unknown', execution_closure='unknown'))
            i = end + 1
            continue
        if token[4] != 'literal':
            if token[0] == '{': depth += 1
            if token[0] == '}': depth -= 1
            if depth < 0: raise ValueError('unbalanced closing brace')
        i += 1
    if depth: raise ValueError('unbalanced top-level braces')
    return result
