// Real installed-source CLI, manager, Raft, scheduler and worker agent. Only the
// task lifecycle is a contract fixture; no TerraBase binary or production mesh.
type coldCLIExecutor struct { mu sync.Mutex; tasks map[string]*coldCLITask }
type coldCLITask struct { done chan struct{}; once sync.Once }
func (*coldCLIExecutor) Describe(context.Context) (*api.NodeDescription,error) {
 return &api.NodeDescription{Platform:&api.Platform{OS:"linux",Architecture:runtime.GOARCH},Resources:&api.Resources{NanoCPUs:8e9,MemoryBytes:8<<30,Generic:[]*api.GenericResource{genericresource.NewDiscrete(swarmruntime.DiskResource,32<<30)}}},nil
}
func (*coldCLIExecutor) Configure(context.Context,*api.Node) error{return nil}
func (*coldCLIExecutor) SetNetworkBootstrapKeys([]*api.EncryptionKey)error{return nil}
func(e *coldCLIExecutor)Controller(task *api.Task)(exec.Controller,error){e.mu.Lock();defer e.mu.Unlock();c:=e.tasks[task.ID];if c==nil{c=&coldCLITask{done:make(chan struct{})};e.tasks[task.ID]=c};return c,nil}
func(*coldCLITask)Update(context.Context,*api.Task)error{return nil}
func(*coldCLITask)Prepare(context.Context)error{return nil}
func(*coldCLITask)Start(context.Context)error{return nil}
func(c *coldCLITask)Wait(ctx context.Context)error{select{case <-ctx.Done():return ctx.Err();case <-c.done:return nil}}
func(c *coldCLITask)Shutdown(context.Context)error{c.once.Do(func(){close(c.done)});return nil}
func(c *coldCLITask)Terminate(ctx context.Context)error{return c.Shutdown(ctx)}
func(c *coldCLITask)Remove(ctx context.Context)error{return c.Shutdown(ctx)}
func(*coldCLITask)Close()error{return nil}
